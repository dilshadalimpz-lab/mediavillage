        let isScanning = true;

        async function onScanSuccess(decodedText) {
            if (!isScanning) return;
            isScanning = false;

            const resultBox = document.getElementById("result");
            resultBox.innerText = "Checking ticket...";
            resultBox.className = "";

            try {
                const docRef = doc(db, "registrations", decodedText);
                const docSnap = await getDoc(docRef);

                if (docSnap.exists()) {
                    const data = docSnap.data();
                    if (data.checkedIn) {
                        resultBox.innerText = `⚠️ ALREADY SCANNED!\n${data.name} (${data.batch})`;
                        resultBox.className = "warning";
                    } else {
                        await updateDoc(docRef, { checkedIn: true });
                        resultBox.innerText = `✅ VERIFIED!\nWelcome ${data.name} (${data.batch})`;
                        resultBox.className = "success";
                    }
                } else {
                    resultBox.innerText = "❌ Invalid Ticket!";
                    resultBox.className = "warning";
                }
            } catch (e) {
                resultBox.innerText = "Error: " + e.message;
            }

            // 3 സെക്കൻഡിന് ശേഷം അടുത്ത സ്കാനിങ്ങിന് തയ്യാറാകുന്നു
            setTimeout(() => { isScanning = true; }, 3000);
        }

        const html5QrcodeScanner = new Html5QrcodeScanner("reader", { fps: 10, qrbox: 250 });
        html5QrcodeScanner.render(onScanSuccess);
